## 🎯 关键代码模式

### API调用模式
```typescript
const loadData = async () => {
  loading.value = true
  try {
    const res = await getXXX({ page, page_size, ...filters })
    dataList.value = res.data.data.list
    total.value = res.data.data.total
  } catch (error) {
    console.error('加载失败:', error)
    alert('加载失败，请重试')
  } finally {
    loading.value = false
  }
}
```

### 表单提交模式
```typescript
const handleSubmit = async () => {
  try {
    if (editingItem.value) {
      await updateXXX(editingItem.value.id, formData.value)
    } else {
      await createXXX(formData.value)
    }
    alert('操作成功')
    closeDialog()
    loadData()
  } catch (error: any) {
    alert(error.response?.data?.msg || '操作失败')
  }
}
```
